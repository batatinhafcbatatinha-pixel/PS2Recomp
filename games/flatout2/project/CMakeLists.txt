# Functions outside the Ghidra map (scripts/gen_gap_functions.py); installed into empty table slots by main.cpp.
# Compiled in batches of 64 per translation unit: 1,174 separate /Od compiles ran out of memory (2026-10-26).
file(GLOB FO2_GAP_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gap-functions/gap_*.cpp")
list(SORT FO2_GAP_FILES)
set(FO2_GAP_SOURCES "")
set(batch "")
set(batch_index 0)
list(LENGTH FO2_GAP_FILES gap_count)
set(gap_seen 0)
foreach(gap_file IN LISTS FO2_GAP_FILES)
    string(APPEND batch "#include \"${gap_file}\"
")
    math(EXPR gap_seen "${gap_seen} + 1")
    math(EXPR batch_fill "${gap_seen} % 64")
    if(batch_fill EQUAL 0 OR gap_seen EQUAL gap_count)
        set(unit "${CMAKE_CURRENT_BINARY_DIR}/gap-units/gap_unit_${batch_index}.cpp")
        file(CONFIGURE OUTPUT "${unit}" CONTENT "${batch}" @ONLY)
        list(APPEND FO2_GAP_SOURCES "${unit}")
        set(batch "")
        math(EXPR batch_index "${batch_index} + 1")
    endif()
endforeach()
target_sources(ps2EntryRunner PRIVATE ${FO2_GAP_SOURCES})
if(MSVC)
    set_source_files_properties(${FO2_GAP_SOURCES} PROPERTIES COMPILE_OPTIONS "/Od;/Ob0;/fp:precise;/bigobj")
else()
    set_source_files_properties(${FO2_GAP_SOURCES} TARGET_DIRECTORY ps2EntryRunner
        PROPERTIES COMPILE_OPTIONS "-O0;-ffp-contract=off")
endif()

# Large resumable switches make whole-game optimizing compiles expensive.
# Bootstrap native code without optimizer expansion; optimize after coverage
# and dispatch integration stabilize. Other projects/runtime retain Release.
option(FO2_OPTIMIZE_AOT "Optimize the large generated game-code library" ON)
if(MSVC)
    # Unoptimized AOT emits large native frames; the observed boot exhausted 1 MiB.
    target_link_options(ps2EntryRunner PRIVATE /STACK:16777216 /MAP)
    if(FO2_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE /O1 /Ob1 /fp:precise)
    else()
        target_compile_options(ps2_game_code PRIVATE /Od /Ob0 /fp:precise)
    endif()
else()
    # GCC/Clang: same policy as MSVC. -ffp-contract=off keeps FMA contraction out (the PS2 FPU has no fused
    # multiply-add); the game thread stack is raised at runtime (PS2X_THREAD_STACK_MB, default 64 MiB).
    if(FO2_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE -O1 -ffp-contract=off)
    else()
        target_compile_options(ps2_game_code PRIVATE -O0 -ffp-contract=off)
    endif()
endif()
