# The native executable of one game, built by tools/ps2port the same way for every game.
#   PORT_DIR    the port folder (game_files/, port.toml)
#   PORT_BUILD  its generated build folder (runner/, chunks/, gap/, port-gen/, iop_aot/, vu1n/)
#   PORT_PROFILE_HANDLERS  optional C++ file of the game's own handlers (a profile in games/<game>)
cmake_minimum_required(VERSION 3.25)
project(PS2PortNative LANGUAGES C CXX)

if(NOT PORT_DIR OR NOT PORT_BUILD)
    message(FATAL_ERROR "Run through tools/ps2port/ps2port.py (PORT_DIR and PORT_BUILD are required)")
endif()
get_filename_component(PORT_PS2RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)

set(PS2X_RUNNER_DIR "${PORT_BUILD}/runner" CACHE PATH "Generated sources" FORCE)
set(PS2X_GAME_CHUNK_DIR "${PORT_BUILD}/chunks" CACHE PATH "Chunk files" FORCE)
set(PS2X_ENABLE_CODWAW_SUPPLEMENTS OFF CACHE BOOL "No other game's address bindings" FORCE)
# macOS stops at OpenGL 4.1, so the GPU rasterizer cannot run there: the software GS is the renderer.
if(APPLE)
    set(PS2X_REQUIRE_GPU_RASTERIZER OFF CACHE BOOL "" FORCE)
    set(PS2X_GPU_COMPUTE_SCANOUT OFF CACHE BOOL "" FORCE)
    set(PS2X_GPU_TEXTURE_HANDOFF OFF CACHE BOOL "" FORCE)
else()
    set(PS2X_REQUIRE_GPU_RASTERIZER OFF CACHE BOOL "Fall back to the software GS without OpenGL 4.6" FORCE)
    set(PS2X_GPU_COMPUTE_SCANOUT ON CACHE BOOL "GPU detile, CRT blend and field scanout" FORCE)
    set(PS2X_GPU_TEXTURE_HANDOFF ON CACHE BOOL "Shared GL texture presentation" FORCE)
endif()
set(PS2X_ENABLE_LTCG OFF CACHE BOOL "Bound compilation memory" FORCE)
set(PS2X_FAST_LINK ON CACHE BOOL "Incremental linking" FORCE)
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "" FORCE)
set(PS2X_SPLIT_GAME_CODE ON CACHE BOOL "Generated code in ps2_game_code" FORCE)
set(PS2X_ENABLE_AGRESSIVE_LOGS OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "" FORCE)

add_subdirectory("${PORT_PS2RECOMP_DIR}" ps2recomp)

# The shared frontend main.cpp is wrapped by this project's main.cpp (it installs the port's tables first).
get_target_property(PORT_ENTRY_SOURCES ps2EntryRunner SOURCES)
list(REMOVE_ITEM PORT_ENTRY_SOURCES "${PORT_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES ${PORT_ENTRY_SOURCES})
target_sources(ps2EntryRunner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
target_include_directories(ps2EntryRunner PRIVATE "${PORT_BUILD}/port-gen")
target_compile_definitions(ps2EntryRunner PRIVATE
    "PS2X_SHARED_MAIN_CPP=\"${PORT_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp\"")
if(PORT_PROFILE_HANDLERS)
    target_compile_definitions(ps2EntryRunner PRIVATE "PORT_PROFILE_HANDLERS=\"${PORT_PROFILE_HANDLERS}\"")
endif()

# Gap functions: compiled 64 per translation unit (one unit each ran the compiler out of memory).
file(GLOB PORT_GAP_FILES CONFIGURE_DEPENDS "${PORT_BUILD}/gap/gap_*.cpp")
list(SORT PORT_GAP_FILES)
set(PORT_GAP_SOURCES "")
set(PORT_GAP_PRELUDE "#include \"${PORT_BUILD}/gap/gap-declarations.h\"\n")
set(batch "${PORT_GAP_PRELUDE}")
set(batch_index 0)
list(LENGTH PORT_GAP_FILES gap_count)
set(gap_seen 0)
foreach(gap_file IN LISTS PORT_GAP_FILES)
    string(APPEND batch "#include \"${gap_file}\"\n")
    math(EXPR gap_seen "${gap_seen} + 1")
    math(EXPR batch_fill "${gap_seen} % 64")
    if(batch_fill EQUAL 0 OR gap_seen EQUAL gap_count)
        set(unit "${CMAKE_CURRENT_BINARY_DIR}/gap-units/gap_unit_${batch_index}.cpp")
        file(CONFIGURE OUTPUT "${unit}" CONTENT "${batch}" @ONLY)
        list(APPEND PORT_GAP_SOURCES "${unit}")
        set(batch "${PORT_GAP_PRELUDE}")
        math(EXPR batch_index "${batch_index} + 1")
    endif()
endforeach()
target_sources(ps2EntryRunner PRIVATE ${PORT_GAP_SOURCES})

if(MSVC)
    set_source_files_properties(${PORT_GAP_SOURCES} PROPERTIES COMPILE_OPTIONS "/Od;/Ob0;/fp:precise;/bigobj")
    target_compile_options(ps2_game_code PRIVATE /O1 /Ob1 /fp:precise)
    # Large native frames of the translated code.
    target_link_options(ps2EntryRunner PRIVATE /STACK:16777216)
    target_compile_options(ps2_runtime PRIVATE /Z7)
    target_compile_options(ps2EntryRunner PRIVATE /Z7)
    target_link_options(ps2EntryRunner PRIVATE /DEBUG:FULL)
else()
    # No FMA contraction: the PS2 FPU has no fused multiply-add.
    set_source_files_properties(${PORT_GAP_SOURCES} TARGET_DIRECTORY ps2EntryRunner PROPERTIES COMPILE_OPTIONS "-O0;-ffp-contract=off")
    target_compile_options(ps2_game_code PRIVATE -O1 -ffp-contract=off)
endif()

# Optical reads come from the extracted game_files folder (port-installer's layout index), never from an ISO.
set(PORT_GAME_ROOT "${PORT_DIR}" CACHE PATH "Folder holding game_files" FORCE)
set(PORT_INSTALLER_DIR "${PORT_PS2RECOMP_DIR}/tools/port-installer" CACHE PATH "port-installer")
include("${PORT_INSTALLER_DIR}/runtime/extracted_files.cmake")

# IOP modules recompiled statically (ps2xIOP/tools/irx_aot.py); each file registers itself with the IOP.
file(GLOB PORT_IOP_AOT_SOURCES CONFIGURE_DEPENDS "${PORT_BUILD}/iop_aot/aot_*.cpp")
target_sources(ps2EntryRunner PRIVATE ${PORT_IOP_AOT_SOURCES})
if(MSVC)
    set_source_files_properties(${PORT_IOP_AOT_SOURCES} PROPERTIES COMPILE_OPTIONS "/O2;/bigobj")
else()
    set_source_files_properties(${PORT_IOP_AOT_SOURCES} TARGET_DIRECTORY ps2EntryRunner PROPERTIES COMPILE_OPTIONS "-O2")
endif()
